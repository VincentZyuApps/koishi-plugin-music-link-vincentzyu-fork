# QQ 官方 Bot API — 群文件上传（分片上传 + 自定义文件名）

## 概述

QQ 官方 Bot API 上传文件有两种方式：

| 方式 | API | 适用场景 |
|------|-----|---------|
| **整体上传** | `POST /v2/groups/{id}/files` | 小文件，base64 塞进请求体 |
| **分片上传** | `upload_prepare` → `upload_part_finish` → `files` | 大文件，更稳定 |

> 整体上传的问题：文件 base64 后膨胀 33%，大文件容易超时或被网关拒绝（500 错误）。
> 分片上传的优势：数据直接 PUT 到腾讯 COS，不经过 API 网关，单片失败只重传单片。

## 文件大小限制（腾讯平台限制）

| 文件类型 | file_type | 上限 |
|---------|-----------|------|
| 图片 | 1 | 30 MB |
| 视频 | 2 | 100 MB |
| 语音 | 3 | 20 MB |
| 文件 | 4 | 100 MB |

---

## 方式一：整体上传（简单但不推荐大文件）

```javascript
const crypto = require('node:crypto');

const APP_ID = '你的AppID';
const CLIENT_SECRET = '你的ClientSecret';
const GROUP_OPENID = '群的openid';
const API_BASE = 'https://api.sgroup.qq.com';

// 获取 access_token
async function getAccessToken() {
  const res = await fetch('https://bots.qq.com/app/getAppAccessToken', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ appId: APP_ID, clientSecret: CLIENT_SECRET }),
  });
  const data = await res.json();
  return data.access_token;
}

// 整体上传文件 + 发送消息
async function uploadAndSendFile(filePath, fileName, accessToken) {
  const fs = require('node:fs');
  const fileBuffer = fs.readFileSync(filePath);
  const base64Data = fileBuffer.toString('base64');

  // Step 1: 上传文件（整体上传，base64 塞进请求体）
  const uploadRes = await fetch(`${API_BASE}/v2/groups/${GROUP_OPENID}/files`, {
    method: 'POST',
    headers: {
      'Authorization': `QQBot ${accessToken}`,
      'X-Union-Appid': APP_ID,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      file_type: 4,           // 4=文件, 1=图片, 2=视频, 3=语音
      srv_send_msg: false,    // false=只上传不发消息，true=上传并直接发送
      file_data: base64Data,  // 整个文件的 base64
    }),
  });
  const fileResult = await uploadRes.json();
  // fileResult = { file_uuid, file_info, ttl }

  console.log('上传成功:', fileResult.file_uuid);

  // Step 2: 发送消息（msg_type=7 媒体消息）
  // ⚠️ 注意：不要带 content 字段（哪怕是空字符串），否则会渲染出一个空行！
  const sendRes = await fetch(`${API_BASE}/v2/groups/${GROUP_OPENID}/messages`, {
    method: 'POST',
    headers: {
      'Authorization': `QQBot ${accessToken}`,
      'X-Union-Appid': APP_ID,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      msg_type: 7,
      media: fileResult,      // { file_uuid, file_info, ttl }
      // 如果需要被动回复，加上 msg_id 和 msg_seq
    }),
  });
  const msgResult = await sendRes.json();
  console.log('消息发送成功:', msgResult.id);
}
```

> ⚠️ 整体上传**不支持自定义文件名**！腾讯 API 的 `file_data` 模式没有 `file_name` 字段。
> 如果需要自定义文件名，必须使用分片上传。

## 方式二：分片上传（推荐，支持自定义文件名）

### 流程

```
Step 1: POST /v2/groups/{id}/upload_prepare
        body: { file_type, file_name, file_size, md5, sha1, md5_10m }
        → 返回 { upload_id, block_size, parts: [{index, presigned_url}], concurrency }

Step 2: 对每个 part（可并发）:
        a. 从文件 Buffer 中切出对应区间的数据
        b. PUT 到 presigned_url（腾讯 COS 预签名链接，不需要 auth header）
        c. POST /v2/groups/{id}/upload_part_finish
           body: { upload_id, part_index, block_size, md5 }

Step 3: POST /v2/groups/{id}/files
        body: { upload_id }
        → 返回 { file_uuid, file_info, ttl }

Step 4: POST /v2/groups/{id}/messages
        body: { msg_type: 7, media: { file_info } }
        → 发送文件消息
```

### 完整代码

```javascript
const crypto = require('node:crypto');
const fs = require('node:fs');

const APP_ID = '你的AppID';
const CLIENT_SECRET = '你的ClientSecret';
const GROUP_OPENID = '群的openid';
const API_BASE = 'https://api.sgroup.qq.com';

// ============ 配置 ============
const MD5_10M_SIZE = 10002432;           // 前 10002432 字节用于 md5_10m 计算
const PART_UPLOAD_TIMEOUT = 300_000;     // 单片上传超时 5 分钟
const PART_UPLOAD_MAX_RETRIES = 2;       // 单片上传最大重试次数
const MAX_CONCURRENT_PARTS = 10;         // 分片并发上限

// ============ 获取 access_token ============
async function getAccessToken() {
  const res = await fetch('https://bots.qq.com/app/getAppAccessToken', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ appId: APP_ID, clientSecret: CLIENT_SECRET }),
  });
  const data = await res.json();
  return data.access_token;
}

// ============ API 请求封装 ============
async function apiRequest(accessToken, method, path, body) {
  const url = `${API_BASE}${path}`;
  const res = await fetch(url, {
    method,
    headers: {
      'Authorization': `QQBot ${accessToken}`,
      'X-Union-Appid': APP_ID,
      'Content-Type': 'application/json',
    },
    body: body ? JSON.stringify(body) : undefined,
  });
  if (!res.ok) {
    const text = await res.text();
    throw new Error(`API ${method} ${path} failed: ${res.status} ${text}`);
  }
  return res.json();
}

// ============ 计算文件哈希 ============
function computeFileHashes(buffer) {
  const md5 = crypto.createHash('md5').update(buffer).digest('hex');
  const sha1 = crypto.createHash('sha1').update(buffer).digest('hex');
  // md5_10m: 文件 > 10002432 字节时，取前 10002432 字节算 MD5；否则等于整文件 MD5
  const md5_10m = buffer.length > MD5_10M_SIZE
    ? crypto.createHash('md5').update(buffer.subarray(0, MD5_10M_SIZE)).digest('hex')
    : md5;
  return { md5, sha1, md5_10m };
}

// ============ PUT 分片数据到 COS 预签名 URL ============
async function putToPresignedUrl(presignedUrl, data, partIndex, totalParts) {
  for (let attempt = 0; attempt <= PART_UPLOAD_MAX_RETRIES; attempt++) {
    try {
      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), PART_UPLOAD_TIMEOUT);
      try {
        const response = await fetch(presignedUrl, {
          method: 'PUT',
          body: data,
          headers: { 'Content-Length': String(data.length) },
          signal: controller.signal,
        });
        if (!response.ok) {
          const body = await response.text().catch(() => '');
          throw new Error(`COS PUT failed: ${response.status} - ${body}`);
        }
        console.log(`  分片 ${partIndex}/${totalParts}: 上传成功 (${data.length} bytes)`);
        return;
      } finally {
        clearTimeout(timeoutId);
      }
    } catch (err) {
      if (attempt < PART_UPLOAD_MAX_RETRIES) {
        const delay = 1000 * Math.pow(2, attempt);
        console.warn(`  分片 ${partIndex}/${totalParts}: 失败，${delay}ms 后重试...`);
        await new Promise(r => setTimeout(r, delay));
      } else {
        throw err;
      }
    }
  }
}

// ============ 通知平台分片完成（带重试） ============
async function uploadPartFinish(accessToken, uploadId, partIndex, blockSize, md5Hex) {
  for (let attempt = 0; attempt <= 2; attempt++) {
    try {
      // 每次重新获取 token，避免长时间上传后 token 过期
      const token = await getAccessToken();
      await apiRequest(token, 'POST', `/v2/groups/${GROUP_OPENID}/upload_part_finish`, {
        upload_id: uploadId,
        part_index: partIndex,
        block_size: blockSize,
        md5: md5Hex,
      });
      return;
    } catch (err) {
      // 错误码 40093001 = 平台异步处理未就绪，需要持续重试
      const bizCode = err.message.match(/(\d{8})/)?.[1];
      if (bizCode === '40093001') {
        console.log(`  分片 ${partIndex}: 平台处理中，1s 后重试...`);
        await new Promise(r => setTimeout(r, 1000));
        continue;
      }
      if (attempt < 2) {
        await new Promise(r => setTimeout(r, 1000 * Math.pow(2, attempt)));
      } else {
        throw err;
      }
    }
  }
}

// ============ 分片上传主函数 ============
async function chunkedUploadFile(filePath, fileName, fileType = 4) {
  const fileBuffer = fs.readFileSync(filePath);
  const fileSize = fileBuffer.length;

  console.log(`\n📤 开始分片上传: ${fileName} (${(fileSize / 1024 / 1024).toFixed(1)}MB)`);

  // Step 1: 计算文件哈希
  const hashes = computeFileHashes(fileBuffer);
  console.log(`  哈希: md5=${hashes.md5.slice(0, 8)}..., sha1=${hashes.sha1.slice(0, 8)}...`);

  // Step 2: 申请上传 → 获取 upload_id + 分片预签名链接
  const accessToken = await getAccessToken();
  const prepareResp = await apiRequest(accessToken, 'POST', `/v2/groups/${GROUP_OPENID}/upload_prepare`, {
    file_type: fileType,       // 1=图片, 2=视频, 3=语音, 4=文件
    file_name: fileName,       // ⭐ 自定义文件名在这里传入！
    file_size: fileSize,
    md5: hashes.md5,
    sha1: hashes.sha1,
    md5_10m: hashes.md5_10m,
  });

  const { upload_id, parts } = prepareResp;
  const block_size = Number(prepareResp.block_size);
  const concurrency = Math.min(Number(prepareResp.concurrency || 1), MAX_CONCURRENT_PARTS);

  console.log(`  upload_id=${upload_id}`);
  console.log(`  block_size=${(block_size / 1024 / 1024).toFixed(1)}MB, 分片数=${parts.length}, 并发=${concurrency}`);

  // Step 3: 逐片上传（支持并发）
  let completedParts = 0;

  async function uploadPart(part) {
    const partIndex = part.index;  // 1-based
    const offset = (partIndex - 1) * block_size;
    const length = Math.min(block_size, fileSize - offset);
    const partBuffer = fileBuffer.subarray(offset, offset + length);
    const md5Hex = crypto.createHash('md5').update(partBuffer).digest('hex');

    // a. PUT 到 COS 预签名 URL（不需要 auth header）
    await putToPresignedUrl(part.presigned_url, partBuffer, partIndex, parts.length);

    // b. 通知平台该分片已完成
    await uploadPartFinish(await getAccessToken(), upload_id, partIndex, length, md5Hex);

    completedParts++;
    console.log(`  进度: ${completedParts}/${parts.length} (${((completedParts / parts.length) * 100).toFixed(0)}%)`);
  }

  // 并发控制：每批最多 concurrency 个分片同时上传
  for (let i = 0; i < parts.length; i += concurrency) {
    const batch = parts.slice(i, i + concurrency);
    await Promise.all(batch.map(part => uploadPart(part)));
  }

  // Step 4: 完成上传 → 获取 file_info
  console.log('  所有分片上传完成，正在完成上传...');
  const finalToken = await getAccessToken();
  const result = await apiRequest(finalToken, 'POST', `/v2/groups/${GROUP_OPENID}/files`, {
    upload_id: upload_id,
  });

  console.log(`✅ 上传完成! file_uuid=${result.file_uuid}, ttl=${result.ttl}s`);
  // result = { file_uuid, file_info, ttl }
  return result;
}

// ============ 发送文件消息 ============
async function sendFileMessage(fileInfo, msgId) {
  const accessToken = await getAccessToken();

  // ⚠️ 不要带 content 字段！否则会渲染出一个空行（腾讯的 bug）
  const body = {
    msg_type: 7,
    media: { file_info: fileInfo },
  };
  // 被动回复需要 msg_id
  if (msgId) {
    body.msg_id = msgId;
    body.msg_seq = 1;
  }

  const res = await apiRequest(accessToken, 'POST', `/v2/groups/${GROUP_OPENID}/messages`, body);
  console.log(`📨 消息发送成功: id=${res.id}`);
  return res;
}

// ============ 使用示例 ============
async function main() {
  // 上传文件（自定义文件名）
  const uploadResult = await chunkedUploadFile(
    './test_kafu.mp3',           // 本地文件路径
    '可不-愛するように.flac',     // ⭐ 自定义文件名（支持中文！）
    4,                           // file_type: 4=文件
  );

  // 发送文件消息
  await sendFileMessage(uploadResult.file_info);
}

main().catch(console.error);
```

## 关键注意事项

### 1. 文件名

- **整体上传**（`file_data` 模式）：不支持自定义文件名，腾讯会随机生成
- **分片上传**（`upload_prepare` 的 `file_name` 字段）：支持自定义文件名，支持中文

### 2. 空行 Bug

发送 `msg_type=7` 媒体消息时，**不要带 `content` 字段**（哪怕是空字符串 `""` 或空格 `" "`）。
带了就会在消息下方渲染出一个空行。这是腾讯 API 的 bug，文档说 `content` 必填但实际不能带。

```javascript
// ✅ 正确
{ msg_type: 7, media: { file_info: "..." } }

// ❌ 错误（会多一个空行）
{ msg_type: 7, media: { file_info: "..." }, content: "" }
{ msg_type: 7, media: { file_info: "..." }, content: " " }
```

### 3. COS 预签名 URL

分片数据 PUT 到的是腾讯 COS 的预签名 URL，**不需要** `Authorization` header。
加了反而可能报错。只需要 `Content-Length`。

### 4. Token 刷新

access_token 有效期约 2 小时。大文件上传耗时长，每个 `upload_part_finish` 调用前建议重新获取 token。

### 5. 错误码 40093001

`upload_part_finish` 可能返回 `40093001`，表示平台异步处理未就绪。需要以 1 秒间隔持续重试，直到成功或超时。

### 6. 每日上传限制

`upload_prepare` 可能返回 `40093002`，表示当日累计上传量超限（约 2GB/天）。此时只能等第二天或让用户手动下载。

## 单聊（C2C）场景

把 `/v2/groups/{group_openid}/` 替换为 `/v2/users/{openid}/` 即可：

```
POST /v2/users/{openid}/upload_prepare
POST /v2/users/{openid}/upload_part_finish
POST /v2/users/{openid}/files          (完成上传)
POST /v2/users/{openid}/messages       (发送消息)
```
